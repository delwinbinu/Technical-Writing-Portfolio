 \+[200\texorpdfstring{$\sim$}{\string~}\+Doxygen extracts comments from source code files to generate structured HTML and PDF documentation.\hypertarget{md__r_e_a_d_m_e_autotoc_md0}{}\doxysubsection{\texorpdfstring{Common Documentation Tags}{Common Documentation Tags}}\label{md__r_e_a_d_m_e_autotoc_md0}

\begin{DoxyItemize}
\item {\ttfamily @file}: Designates the file name and general description.
\item {\ttfamily @brief}: Single-\/line summary of what a function or module does.
\item {\ttfamily @param}: Describes an input argument and its purpose.
\item {\ttfamily @return}: Explains the return value and error states.
\end{DoxyItemize}\hypertarget{md__r_e_a_d_m_e_autotoc_md1}{}\doxysubsection{\texorpdfstring{How to Build the Documentation}{How to Build the Documentation}}\label{md__r_e_a_d_m_e_autotoc_md1}

\begin{DoxyEnumerate}
\item Generate configuration template: {\ttfamily doxygen -\/g Doxyfile}
\item Run compilation: {\ttfamily doxygen Doxyfile}
\item View output by opening {\ttfamily html/\+index.\+html} in any web browser. 
\end{DoxyEnumerate}
